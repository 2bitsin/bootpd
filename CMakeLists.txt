cmake_minimum_required(VERSION 3.20)
project(bootpd VERSION 1.1.0 DESCRIPTION "DHCP/BOOTP and TFTP server for network booting" LANGUAGES CXX)

option(BOOTPD_BUILD_TESTS "Build the unit tests (requires GoogleTest)" OFF)
option(BOOTPD_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Common compiler settings, applied to every target through this interface library.
add_library(bootpd_build_options INTERFACE)
target_compile_features(bootpd_build_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(bootpd_build_options INTERFACE /W4 /permissive- /utf-8 /Zc:__cplusplus)
  target_compile_definitions(bootpd_build_options INTERFACE _CRT_SECURE_NO_WARNINGS)
  if(BOOTPD_WARNINGS_AS_ERRORS)
    target_compile_options(bootpd_build_options INTERFACE /WX)
  endif()
else()
  target_compile_options(bootpd_build_options INTERFACE -Wall -Wextra -Wpedantic)
  if(BOOTPD_WARNINGS_AS_ERRORS)
    target_compile_options(bootpd_build_options INTERFACE -Werror)
  endif()
endif()

add_subdirectory(common)
add_subdirectory(bootpd)

if(BOOTPD_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

include(GNUInstallDirs)
install(TARGETS bootpd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES example/config.ini DESTINATION ${CMAKE_INSTALL_DATADIR}/bootpd/example)
if(WIN32)
  install(FILES example/bootpd.bat DESTINATION ${CMAKE_INSTALL_DATADIR}/bootpd/example)
endif()
